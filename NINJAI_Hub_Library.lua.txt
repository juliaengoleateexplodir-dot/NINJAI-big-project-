local function trim(s) return (tostring(s):match("^%s*(.-)%s*$")) end
local function pct(x) return string.format("%.0f%%", (x or 0) * 100) end
local function clip(s, n) s = tostring(s):gsub("%s+", " "); if #s > n then return s:sub(1, n - 1) .. "…" end return s end
local function fmtTime(t) return os.date("%H:%M:%S", t) end
local function row(text, value) return { Text = tostring(text), Value = tostring(value) } end
local function meta() return NN.VersionMeta(NN.ActiveVersion) end
local function copy(text)
	local fn = rawget(_G, "setclipboard") or rawget(_G, "toclipboard") or (getgenv and getgenv().setclipboard)
	if type(fn) == "function" then local ok = pcall(fn, text); return ok end
	return false
end
local function stateText()
	local R = NN.Run
	if R.running then return R.paused and "em pausa" or string.format("treinando (%s) época %d/%d", R.kind, R.epoch, R.total) end
	return "ocioso"
end

-- executa uma ação com permissão, auditoria (mesmo registro do Hub) e aviso do resultado
local function act(label, level, fn, bypass)
	return function()
		if not HUB.Can(level) then
			notify("✖ Sem permissão para \"" .. label .. "\" (precisa de " .. level .. "; você está como " .. HUB.role .. ")")
			return
		end
		if level == "admin" and not bypass and HUB.Auth.protect and HUB.Auth.admin and (HUB.editUntil or 0) < os.time() then
			notify("Configurações protegidas: libere as edições na aba Segurança (PIN)")
			return
		end
		if level ~= "viewer" then NN.Log("admin", label .. " [" .. HUB.role .. "]") end
		local ok, a, b = pcall(fn)
		if not ok then
			NN.Fail("hubui:" .. label, a)
			notify("✖ Erro: " .. clip(a, 140))
		elseif a == false then
			notify("⚠ " .. clip(b or "não foi possível", 160))
		elseif type(a) == "string" then
			notify("✔ " .. clip(a, 160), true)
		elseif type(b) == "string" then
			notify("✔ " .. clip(b, 160), true)
		else
			notify("✔ " .. label, true)
		end
	end
end

-- pergunta com caixa de texto (uma linha ou várias); chama cb(texto) ao confirmar
local function ask(title, placeholder, default, multiline, cb)
	closeKind("ask")
	local f = windowFrame(title, 0.9, multiline and 0.6 or 0.3)
	current.ask = f
	local box = new("TextBox", { PlaceholderText = placeholder or "", Text = default or "", ClearTextOnFocus = false, MultiLine = multiline and true or false, TextWrapped = multiline and true or false,
		Font = multiline and Enum.Font.Code or Enum.Font.Gotham, TextSize = 13, TextColor3 = C.text, PlaceholderColor3 = C.dim, BackgroundColor3 = C.field,
		Size = UDim2.new(1, -24, 1, -90), Position = UDim2.new(0, 12, 0, 38), TextXAlignment = Enum.TextXAlignment.Left, TextYAlignment = multiline and Enum.TextYAlignment.Top or Enum.TextYAlignment.Center,
		ClipsDescendants = true }, f)
	corner(box, 8)
	new("UIPadding", { PaddingLeft = UDim.new(0, 8), PaddingRight = UDim.new(0, 8), PaddingTop = UDim.new(0, 6) }, box)
	local bar = new("Frame", { BackgroundTransparency = 1, Size = UDim2.new(1, -24, 0, 30), Position = UDim2.new(0, 12, 1, -40) }, f)
	new("UIListLayout", { FillDirection = Enum.FillDirection.Horizontal, Padding = UDim.new(0, 8) }, bar)
	local function submit()
		local t = box.Text
		closeKind("ask")
		local ok, err = pcall(cb, t)
		if not ok then NN.Fail("hubui:ask", err); notify("✖ Erro: " .. clip(err, 140)) end
	end
	button(bar, "OK", submit, C.accent)
	button(bar, "Cancelar", function() closeKind("ask") end)
	if not multiline then box.FocusLost:Connect(function(enter) if enter then submit() end end) end
	return box
end

-- botão que pergunta antes de agir: permissão primeiro, depois a caixa, depois a ação
local function askAction(label, level, title, placeholder, default, multiline, handler)
	return function()
		if not HUB.Can(level) then
			notify("✖ Sem permissão para \"" .. label .. "\" (precisa de " .. level .. "; você está como " .. HUB.role .. ")")
			return
		end
		ask(title, placeholder, default, multiline, function(text)
			act(label, level, function() return handler(text) end)()
		end)
	end
end

-- visão 3D (rede, módulos ou conhecimento) usando a cena do Hub do motor
local function view3D(kind)
	closeKind("view")
	local titles = { rede = "Rede neural em 3D", modulos = "Módulos em 3D", conhecimento = "Conhecimento em 3D" }
	local f = windowFrame(titles[kind] or "3D", 0.95, 0.88)
	current.view = f
	local holder = new("Frame", { BackgroundTransparency = 1, Size = UDim2.new(1, -24, 1, -118), Position = UDim2.new(0, 12, 0, 36) }, f)
	local V = HUB.MakeView(holder, 300, 1)
	V.vp.Size = UDim2.new(1, 0, 1, 0)
	local legend = new("TextLabel", { Text = "", Font = Enum.Font.Gotham, TextSize = 11, TextColor3 = C.dim, BackgroundTransparency = 1, TextWrapped = true,
		Size = UDim2.new(1, -24, 0, 36), Position = UDim2.new(0, 12, 1, -80), TextXAlignment = Enum.TextXAlignment.Left, TextYAlignment = Enum.TextYAlignment.Top }, f)
	local mode = "pesos"
	local function redraw()
		local ok, text
		if kind == "rede" then ok, text = pcall(HUB.SceneNetwork, V, mode)
		elseif kind == "modulos" then ok, text = pcall(HUB.SceneModules, V)
		else ok, text = pcall(HUB.SceneKnowledge, V) end
		legend.Text = ok and tostring(text) or ("erro: " .. tostring(text))
	end
	redraw()
	local bar = new("Frame", { BackgroundTransparency = 1, Size = UDim2.new(1, -24, 0, 30), Position = UDim2.new(0, 12, 1, -40) }, f)
	new("UIListLayout", { FillDirection = Enum.FillDirection.Horizontal, Padding = UDim.new(0, 6), Wraps = true }, bar)
	button(bar, "Zoom +", function() V.Zoom(0.8) end)
	button(bar, "Zoom −", function() V.Zoom(1.25) end)
	button(bar, "Girar", function() V.spin = not V.spin end)
	button(bar, "Atualizar", redraw)
	if kind == "rede" then
		button(bar, "Pesos/Fluxo", function() mode = (mode == "pesos") and "fluxo" or "pesos"; redraw() end)
		button(bar, "Testar frase", function()
			ask("Testar frase na rede", "ex.: criar um script de pulo duplo", "", false, function(t)
				local r = NN.Predict(t)
				if not r then notify("Sem rede treinada"); return end
				local p = {}
				for _, x in ipairs(r.top) do p[#p + 1] = x.label .. " " .. pct(x.prob) end
				notify("Previsão: " .. table.concat(p, " · "), true)
				redraw()
			end)
		end)
	end
	button(bar, "Fechar", function() closeKind("view") end, C.accent)
end

-- muda a rede numa cópia, valida, cria ponto de restauração e só então troca (igual ao Hub próprio)
local function commitArch(label, mutate)
	local base = NN.Net
	if not base then return false, "não há rede ativa" end
	local clone = Network.Clone(base)
	local ok, msg = mutate(clone)
	if not ok then return false, msg or "não foi possível" end
	local good, probs = AM.Validate(clone, #clone.labels)
	if not good then return false, "ficaria inválida: " .. tostring(probs[1]) end
	MC.Point("antes: " .. label)
	local data, sum = NN.LastData, { epochs = 0, samples = 0, loss = 0, trainAcc = 0, valAcc = 0 }
	if data then
		local _, ta = Training.Evaluate(clone, data.train)
		local vl, va = Training.Evaluate(clone, #data.val > 0 and data.val or data.train)
		sum = { epochs = 0, samples = #data.train, loss = vl, valLoss = vl, trainAcc = ta, valAcc = va }
	end
	local prevMeta = meta()
	local approved, txt = NN.Adopt(clone, sum, { kind = "manual", force = true, data = data, note = label })
	if not approved then return false, txt end
	local dropped = prevMeta and sum.valAcc < prevMeta.valAcc - 0.05
	if dropped and NN.Settings.autoLearn and not NN.Run.running then
		NN.StartTraining({})
		return true, label .. " — a precisão caiu, retreinando automaticamente"
	end
	return true, label .. (msg and (" — " .. tostring(msg)) or "") .. (dropped and " (retreine para recuperar a precisão)" or "")
end

local function logRows(kinds, n)
	local out, A = {}, NN.Activity
	for i = #A, 1, -1 do
		if #out >= n then break end
		local a = A[i]
		if not kinds or kinds[a.kind] then out[#out + 1] = row(fmtTime(a.t) .. " · " .. a.kind, clip(a.msg, 90)) end
	end
	if #out == 0 then out[1] = row("—", "nada ainda") end
	return out
end

local function urlEncode(s) return (s:gsub("[^%w%-%._~]", function(ch) return string.format("%%%02X", ch:byte()) end)) end
local function urlDecode(s) return (s:gsub("%%(%x%x)", function(h) return string.char(tonumber(h, 16)) end)) end
local function strip(s)
	s = s:gsub("<.->", "")
	s = s:gsub("&amp;", "&"):gsub("&quot;", '"'):gsub("&#x27;", "'"):gsub("&lt;", "<"):gsub("&gt;", ">")
	return trim(s)
end

	local ok, body = M.ExecutorAdapter.HttpGet("https://html.duckduckgo.com/html/?q=" .. urlEncode(q))
	if not ok then return false, "falha na pesquisa: " .. clip(body, 100) end
	local results = {}
	for href, title in body:gmatch('class="result__a"[^>]-href="(.-)"[^>]*>(.-)</a>') do
		local real = href:match("uddg=([^&]+)")
		results[#results + 1] = { title = strip(title), url = real and urlDecode(real) or href }
		if #results >= 6 then break end
	end
	local k = 0
	for snippet in body:gmatch('class="result__snippet"[^>]*>(.-)</a>') do
		k = k + 1
		if results[k] then results[k].snippet = strip(snippet) end
	end
	if #results == 0 then return false, "nenhum resultado" end
	local t = {}
	for i, r in ipairs(results) do t[#t + 1] = i .. ". " .. r.title .. "\n" .. clip(r.snippet or "", 180) .. "\n" .. clip(r.url, 90) .. "\n" end
	showText("Resultados para \"" .. q .. "\"", table.concat(t, "\n") .. "\nDepois de ler, você pode salvar um na base.")

	end)

end)
